cmake_minimum_required(VERSION 3.15)
project(QmClientSetup LANGUAGES CXX RC)

if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "QmClient Setup requires Windows and MSVC")
endif()
if(NOT EXISTS "${QM_SETUP_ENGINE}")
  message(FATAL_ERROR "QM_SETUP_ENGINE must point to the compiled Inno Setup engine")
endif()
if(NOT QM_SETUP_VERSION MATCHES "^[0-9]+\\.[0-9]+(-preview\\.[0-9]+)?$")
  message(FATAL_ERROR "Invalid QM_SETUP_VERSION")
endif()
if(NOT QM_SETUP_APP_ID MATCHES "^[0-9A-Fa-f-]+$")
  message(FATAL_ERROR "Invalid QM_SETUP_APP_ID")
endif()
get_filename_component(QM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(TO_CMAKE_PATH "${QM_SETUP_ENGINE}" QM_SETUP_ENGINE_RC)
set(QM_SETUP_ICON "${QM_ROOT}/qmclient_scripts/installer/assets/setup-icon.ico")
configure_file(setup_build.h.in setup_build.h @ONLY)
configure_file(setup.rc.in setup.rc @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/setup.rc" PROPERTIES OBJECT_DEPENDS "${QM_SETUP_ENGINE}")
add_executable(qmclient-setup WIN32 setup_ui.cpp setup_engine.cpp "${CMAKE_CURRENT_BINARY_DIR}/setup.rc")
target_compile_features(qmclient-setup PRIVATE cxx_std_17)
target_include_directories(qmclient-setup PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(qmclient-setup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(qmclient-setup PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/permissive-;/EHsc>")
target_link_options(qmclient-setup PRIVATE /MANIFEST:NO)
target_link_libraries(qmclient-setup PRIVATE d2d1 dwrite dwmapi windowscodecs ole32 shell32 advapi32 comctl32 uxtheme uuid)
set_target_properties(qmclient-setup PROPERTIES OUTPUT_NAME "QmClient-Setup" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
